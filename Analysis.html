## Release each item one at a time (to estimate effect sizes of each item's DIF, below)
released_models <- map(
  .x = 1:9,
  .f = ~ {
    
    # Fit a model with that item's parameters free
    model_free_.x <- multipleGroup(
      df[, phq_items],
      model = 1,
      group = df$sample_fct,
      itemtype = "graded",
      invariance = c(
        phq_items[-.x], # Free parameters of variable .x
        "free_means",
        "free_var"
      ),
      verbose = F
    )
    
    return(model_free_.x)
    
  }
)


## Use released models to estimate effect sizes of each item's DIF

# SIDS: Signed item difference in the sample (i.e., across HMS respondents); 
#       the average expected (model-implied) difference between HMS and NHANES 
#       item responses at a given theta, across the theta distribution of HMS
# UIDS: Unsigned item difference in the sample; above, but average absolute value
#       at every theta
# ESSD: Expected score standardized difference; SIDS but adjusted for SD of 
#       expected scores (as low-frequency, low-SD items otherwise produce lower 
#       differences)


## Build tibble of expected values (and probabilities of each response level) given group, item, theta
item_group_combinations <- expand_grid(
  item = 1:9,
  group = groups
)

expected_values_inv <- map2(
  .x = item_group_combinations$item,
  .y = item_group_combinations$group,
  .f = ~ {
    
    model_.x <- released_models[[.x]]
    
    item_.x_.y <- extract.item(model_.x, .x, .y)

    expected_values_.x_.y <- tibble(theta = seq(-5, 5, .01)) %>%
      mutate(
        item = .x,
        group = .y,
        ev = expected.item(item_.x_.y, theta),
        p1 = probtrace(item_.x_.y, theta)[, "P.1"],
        p2 = probtrace(item_.x_.y, theta)[, "P.2"],
        p3 = probtrace(item_.x_.y, theta)[, "P.3"],
        p4 = probtrace(item_.x_.y, theta)[, "P.4"]
      )
    
    return(expected_values_.x_.y)
    
  }
) %>%
  bind_rows()

  "No Anchors (Iterative)" = list(ev = expected_values_inv, mod = model_inv),
  "Empirical Anchors (4, 5)" = list(ev = expected_values_emp, mod = model_emp),
  "Theory-Driven Anchors (1, 2)" = list(ev = expected_values_the, mod = model_the)

  imap(~ {
    
    # EAP theta for each HMS respondent
    theta_hat_hms <- fscores(extract.group(.x$mod, "HMS"), method = "EAP")[, 1]

    # TCC gap (HMS minus NHANES) on the theta grid
    tcc_gap <- .x$ev %>%
      group_by(theta, group) %>%
      summarize(total = sum(ev), .groups = "drop") %>%
      pivot_wider(names_from = group, values_from = total) %>%
      mutate(
        theta = round(theta, 2), # For join
        gap = HMS - NHANES
      )

    # Join to actual HMS respondents' thetas, then summarize
    impact_.y <- tibble(theta = round(theta_hat_hms, 2)) %>%
      left_join(tcc_gap, by = "theta") %>%
      summarize(
        scheme = .y,
        STDS = mean(gap),
        UTDS = mean(abs(gap))
      )
    
    return(impact_.y)
    
  }) %>%
  bind_rows()
